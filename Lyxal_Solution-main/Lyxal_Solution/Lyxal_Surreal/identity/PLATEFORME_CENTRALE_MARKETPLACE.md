## 🔧 Architecture Technique - SDK Partenaires

### SDK Lyxal Identity

```typescript
// ============================================
// @lyxal/identity-sdk (NPM Package)
// ============================================

import { LyxalAuth, LyxalAPI } from '@lyxal/identity-sdk';

/**
 * SDK pour les partenaires qui veulent s'intégrer
 * à la plateforme Lyxal Central
 */

// Configuration
const lyxal = new LyxalAuth({
  appId: 'mailchimp_campaign',
  appSecret: process.env.LYXAL_APP_SECRET,
  redirectUri: 'https://mailchimp.com/oauth/lyxal/callback',
  scopes: ['profile', 'email', 'contacts:read', 'campaigns:write']
});

// Route d'authentification
app.get('/login/lyxal', (req, res) => {
  const authUrl = lyxal.getAuthorizationUrl({
    state: req.session.state
  });
  res.redirect(authUrl);
});

// Callback OAuth
app.get('/oauth/lyxal/callback', async (req, res) => {
  const { code, state } = req.query;
  
  try {
    // Échange du code contre un token
    const tokens = await lyxal.exchangeCodeForTokens(code);
    
    // tokens = {
    //   access_token: 'xxx',
    //   refresh_token: 'yyy',
    //   expires_in: 3600,
    //   lyxal_id: 'jean_dupont_123abc',
    //   email: 'jean@restaurant-bistro.fr'
    // }
    
    // Stockage des tokens
    await saveTokens(tokens);
    
    // Utilisateur authentifié via Lyxal !
    res.redirect('/dashboard');
  } catch (error) {
    res.status(500).json({ error: 'Authentication failed' });
  }
});

// Utilisation de l'API Lyxal
const lyxalAPI = new LyxalAPI({
  accessToken: tokens.access_token
});

// Récupération des contacts depuis Lyxal
const contacts = await lyxalAPI.getContacts({
  lyxal_id: tokens.lyxal_id,
  filters: { type: 'customer', status: 'active' }
});

// contacts = [
//   { id: '123', name: 'Client A', email: 'client-a@example.com' },
//   { id: '456', name: 'Client B', email: 'client-b@example.com' }
// ]

// Synchronisation dans Mailchimp
await mailchimp.lists.addMembers(contacts);
```

### API Lyxal pour Partenaires

```typescript
// ============================================
// API ENDPOINTS LYXAL (Backend Lyxal)
// ============================================

// POST /oauth/authorize
// Demande d'autorisation par un partenaire
app.post('/oauth/authorize', async (req, res) => {
  const { app_id, redirect_uri, scopes, state } = req.body;
  
  // Vérification app partenaire
  const app = await db.query(`
    SELECT * FROM partner_apps
    WHERE app_id = $app_id AND status = 'active'
  `, { app_id });
  
  if (!app) {
    return res.status(404).json({ error: 'App not found' });
  }
  
  // Redirection vers page de consentement
  res.json({
    consent_url: `https://auth.lyxal.com/consent?app_id=${app_id}&scopes=${scopes}&state=${state}`
  });
});

// POST /oauth/token
// Échange code contre tokens
app.post('/oauth/token', async (req, res) => {
  const { code, app_id, app_secret } = req.body;
  
  // Vérification credentials
  const app = await verifyAppCredentials(app_id, app_secret);
  
  // Échange code
  const authData = await exchangeCode(code);
  
  // Génération tokens
  const accessToken = generateAccessToken(authData);
  const refreshToken = generateRefreshToken(authData);
  
  res.json({
    access_token: accessToken,
    refresh_token: refreshToken,
    expires_in: 3600,
    token_type: 'Bearer',
    lyxal_id: authData.lyxal_id,
    email: authData.email,
    granted_scopes: authData.granted_scopes
  });
});

// GET /api/v1/contacts
// API pour récupérer contacts (avec permission)
app.get('/api/v1/contacts', authenticateApp, checkPermission('contacts:read'), async (req, res) => {
  const { lyxal_id } = req.app_user;
  
  // Requête cross-namespace sécurisée
  const contacts = await db.query(`
    SELECT * FROM user_saas_contexts
    WHERE lyxal_id = $lyxal_id
  `, { lyxal_id });
  
  res.json({ contacts });
});
```
